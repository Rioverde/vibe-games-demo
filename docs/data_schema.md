# Схема данных боевого пропуска

Proto-подобный набросок структур, которые экран ожидает от сервера.
Синтаксис абстрактный — важна форма данных, а не соответствие генератору.

```
rpc BattlePass {

  // ─── Награда на дорожке ───────────────────────────────────────
  message Reward {
    int32   id                 "id награды";
    lstring name       = 64    "Название, показывается когда награда выбрана";
    lstring icon       = 128   "Ключ ассета предмета";
    int32   amount             "Количество; 1 — пилюля 'xN' не рисуется";
    int32   tier               "Тир карточки 0..3 — задаёт градиент подложки и свечение";
    bool    premium            "Лежит на премиум-дорожке: корона вместо подарка";
    lstring hint       = 160   "Текст для 'i' рядом с названием; пусто — иконки нет";
  }

  // ─── Уровень дорожки ──────────────────────────────────────────
  message Level {
    int32    number                "Номер уровня, подпись на пине под карточкой";
    int32    state                 "0 locked, 1 available, 2 claimed";
    bool     big                   "Финальный уровень — карточка крупнее и с рамкой";
    repeated Reward rewards max 4  "Награды уровня; забираются одной кнопкой";
  }

  // ─── Задание ──────────────────────────────────────────────────
  message Quest {
    int32   id                 "id задания";
    lstring text       = 160   "Условие: 'Используйте предмет (Энергетик) 10 раз'";
    int32   xpReward           "Сколько опыта БП даёт";
    int32   progress           "Текущий прогресс";
    int32   goal               "Цель; progress/goal — дробь и полоска";
    bool    claimed            "Опыт уже забран — кнопка 'Готово'";
    int64   expiresAt          "Unix-время сгорания; 0 — не сгорает";
  }

  // ─── Правый промо-блок ────────────────────────────────────────
  // Один блок с разной начинкой: до покупки продаёт премиум,
  // после — продаёт уровни. Экран не знает, что именно рекламируется.
  message Promo {
    lstring title      = 64    "'Элитный пропуск' / 'Повышение уровня'";
    lstring subtitle   = 192   "Описание под заголовком";
    lstring cta        = 48    "Подпись кнопки: 'Прокачать' / 'Повысить уровень'";
    lstring art        = 128   "Ключ ассета баннера";
    bool    ctaEnabled         "false на макс. уровне — кнопка гаснет";
    lstring ctaDisabledText = 64 "'Достигнут максимальный уровень'";
    bool    claimAll           "Показать 'Забрать все награды' под кнопкой";
  }

  // ─── Пейлоад главного экрана ──────────────────────────────────
  outgoing BattlePassShow {
    int32   seasonId               "id сезона";
    lstring seasonTitle = 64       "'Дай пять!' — заголовок в инфобаре";
    int64   endsAt                 "Unix-время конца сезона; из него таймер '15д 12ч 42м'";

    int32   level                  "Текущий уровень — число в кольце";
    int32   maxLevel               "Последний уровень сезона";
    int32   xp                     "Опыт внутри текущего уровня";
    int32   xpPerLevel             "Опыт до следующего; xp/xpPerLevel — заполнение кольца";

    bool    premiumOwned           "Куплен ли премиум — главный переключатель состояния";
    int64   claimDeadline          "Сезон кончился, но награды ещё можно забрать; 0 — сезон идёт";

    int32   selectedLevel          "Какой уровень раскрыт в центре экрана";
    Promo   promo                  "Правый блок";

    repeated Reward pending max 20  "Награды в лотке у начала дорожки, ждут кнопки «Получи все сразу!»";
    repeated Level  levels  max 100 "Дорожка уровней";
    repeated Quest  quests  max 20  "Задания для второго экрана и для карточки-превью";
  }
}
```

## Решения по схеме

- **`pending` — свой список, а не срез уровней.** В макете пины дорожки идут 1, 2, 3 … без
  пропусков, а у карточек лотка пина нет вовсе — значит они вне последовательности уровней.
  Проверено по кадру: лоток из трёх карточек и плашка нарисованы **и** в состоянии «премиум
  куплен», поэтому он не привязан к покупке. Опустошает его только само действие «Получи
  все сразу!».
- **Состояние экрана выводится из данных, отдельного «режима» нет.**
  `premiumOwned` разводит «премиум не куплен / куплен», `level == maxLevel` — «максимальный
  уровень», `claimDeadline > 0` — «Battle Pass завершён». Экран не может рассинхронизироваться
  с бэкендом, потому что не хранит режим у себя.
- **`Promo` — один блок с подменяемой начинкой.** В макете правая панель до покупки продаёт
  премиум, после — уровни, на максимуме гаснет. Это один и тот же прямоугольник с разным
  текстом, поэтому в схеме это одна структура, а не три ветки.
- **`state` живёт на уровне, а не на награде.** В слоте лежит пачка предметов, но кнопка
  «Забрать» одна, поэтому `locked/available/claimed` — свойство уровня.
- **Времена абсолютные (`endsAt`, `expiresAt`, `claimDeadline`), не «осталось N».**
  Таймер тикает на клиенте и не врёт, если экран повисел в фоне.
- **`xp` считается внутри уровня.** Кольцо прогресса рисуется без обхода всей дорожки.
- **Карточка-превью задания на главной берётся из `quests`**, отдельного дублирующего
  поля нет — иначе два источника правды разъедутся.
- **`repeated` всегда с `max`**, как в образце из задания.
